🎮 Arduino Serial Rock, Paper, Scissors

An interactive Rock, Paper, Scissors game played entirely on an Arduino Uno R3 using your computer's Serial Monitor. Developed as a standalone project requiring zero external components or wiring.

📌 Project Overview

This project turns your Arduino into a local game server. It accepts user input typed directly through the Arduino IDE Serial Monitor, generates a pseudo-random choice for the computer using analog background noise (analogRead(A0)), evaluates the game logic, and triggers celebratory visual feedback using the Arduino's built-in LED (Pin 13) whenever you win!

✨ Features

Zero Hardware Setup: No breadboards, resistors, or jumper wires needed—just your Arduino and a USB cable.

Interactive Serial I/O: Play continuous rounds directly from your computer keyboard.

True Randomness Seed: Uses floating analog pin noise to ensure random computer choices every session.

LED Celebration: Flashes the built-in LED whenever the player claims victory!

🛠️ Hardware Requirements

Arduino Uno R3 (or any standard Arduino board with an onboard LED)

USB Cable (to connect the board to your computer and power it)

⚙️ How to Play

Connect your Arduino board to your computer via USB.

Open the Arduino IDE, paste the source code, and upload it to your board.

Open the Serial Monitor (Ctrl + Shift + M on Windows/Linux or Cmd + Shift + M on macOS).

Set the baud rate dropdown to 9600.

Type your choice and press Enter:

1 = Rock 🪨

2 = Paper 📄

3 = Scissors ✂️

// ==========================================
// Arduino Serial Rock, Paper, Scissors Game
// ==========================================

void setup() {
  // Initialize serial communication at 9600 baud
  Serial.begin(9600);
  
  // Seed the random number generator using analog noise from unconnected Pin A0
  // This ensures the computer makes a different random choice every game.
  randomSeed(analogRead(A0));
  
  // Print welcome message and instructions
  Serial.println("=========================================");
  Serial.println("   WELCOME TO ARDUINO ROCK, PAPER, SCISSORS");
  Serial.println("=========================================");
  Serial.println("Instructions:");
  Serial.println("Type one of the following letters and press Enter:");
  Serial.println("  R or r -> Rock ");
  Serial.println("  P or p -> Paper ");
  Serial.println("  S or s -> Scissors ");
  Serial.println("-----------------------------------------");
  Serial.println("Type your choice to start!");
  Serial.println("");
}

void loop() {
  // Check if data is available in the Serial Monitor input
  if (Serial.available() > 0) {
    // Read the user's input as a character and convert it to uppercase
    char userChoice = toupper(Serial.read());
    
    // Clear any leftover characters in the serial buffer (like newline characters)
    while (Serial.available() > 0) {
      Serial.read();
    }
    
    // Validate user input
    if (userChoice == 'R' || userChoice == 'P' || userChoice == 'S') {
      // Computer makes a random choice: Rock, Paper, or Scissors
      char choices[3] = {'R', 'P', 'S'};
      char computerChoice = choices[random(0, 3)]; 
      
      // Print player's choice
      Serial.print("You chose: ");
      printChoiceName(userChoice);
      
      // Print computer's choice
      Serial.print("Computer chose: ");
      printChoiceName(computerChoice);
      
      // Determine the winner
      determineWinner(userChoice, computerChoice);
      
      Serial.println("-----------------------------------------");
      Serial.println("Play again! Type R, P, or S:");
    } else {
      // Handle invalid input (ignoring accidental newline characters/spaces if any slipped through)
      if (userChoice != '\n' && userChoice != '\r') {
        Serial.println("Invalid choice! Please type R (Rock), P (Paper), or S (Scissors).");
      }
    }
  }
}

// Helper function to print text name of choices
void printChoiceName(char choice) {
  if (choice == 'R') {
    Serial.println("Rock 🪨");
  } else if (choice == 'P') {
    Serial.println("Paper 📄");
  } else if (choice == 'S') {
    Serial.println("Scissors ✂️");
  }
}

// Helper function to calculate game rules and print result
void determineWinner(char player, char computer) {
  if (player == computer) {
    Serial.println("Result: It's a TIE! 🤝");
  } 
  else if ((player == 'R' && computer == 'S') || 
           (player == 'P' && computer == 'R') || 
           (player == 'S' && computer == 'P')) {
    Serial.println("Result: You WIN! 🎉");
    
    // Fun bonus: Flash the built-in LED (Pin 13) 3 times when you win!
    for(int i = 0; i < 3; i++) {
      digitalWrite(LED_BUILTIN, HIGH);
      delay(150);
      digitalWrite(LED_BUILTIN, LOW);
      delay(150);
    }
  } 
  else {
    Serial.println("Result: Computer WINS! 🤖");
  }
}
